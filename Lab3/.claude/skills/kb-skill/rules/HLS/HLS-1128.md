```cpp
typedef ap_axiu<32, 1, 1, 1> beat_t;
void scale(hls::stream<beat_t>& src, hls::stream<beat_t>& dst) {
#pragma HLS INTERFACE mode=axis port=src
#pragma HLS INTERFACE mode=axis port=dst
#pragma HLS INTERFACE mode=s_axilite port=return
  bool done = false;
  while (!done) {
#pragma HLS PIPELINE II=1
    beat_t b = src.read(), o;
    o.data = b.data << 1; o.last = b.last; o.keep = -1;
    dst.write(o);
    done = b.last;
  }
}
```
