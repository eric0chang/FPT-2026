```cpp
void fill_lut(short lut[LUT_N]) {
  for (int k = 0; k < LUT_N; k++)
    lut[k] = (short)(16384.0 * sin(M_PI * k / LUT_N));
}
int scale(short x, unsigned char k) {
  short lut[LUT_N];
  fill_lut(lut);
  return x * lut[k];
}
```
